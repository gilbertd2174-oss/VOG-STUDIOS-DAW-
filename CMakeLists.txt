cmake_minimum_required(VERSION 3.22)
project(VOGStudio VERSION 0.2.0 LANGUAGES C CXX)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.8 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_gui_app(VOGStudio PRODUCT_NAME "VOG Studio" COMPANY_NAME "Davoe Instruments" BUNDLE_ID com.davoe.vogstudio)
target_sources(VOGStudio PRIVATE Source/Main.cpp)
target_compile_features(VOGStudio PRIVATE cxx_std_17)
target_compile_definitions(VOGStudio PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_AU=$<BOOL:${APPLE}> JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(VOGStudio PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
